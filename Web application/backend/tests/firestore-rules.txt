// Optimized Firestore Security Rules
// Copy this to your Firebase Console -> Firestore -> Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Collection des utilisateurs - seul l'utilisateur peut accéder à ses données
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Collection publique - lecture libre, écriture pour utilisateurs connectés
    match /public/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Collection de test - accès libre pour les tests (TEMPORAIRE)
    match /test/{docId} {
      allow read, write: if true;
    }

    // Collections système - accès admin seulement
    match /admin/{docId} {
      allow read, write: if request.auth != null && 
        request.auth.token.admin == true;
    }

    // Règle par défaut - tout refuser sauf si explicitement autorisé
    // Note: Cette règle sera appliquée en dernier
    match /{document=**} {
      allow read, write: if false;  // Changé de 'true' à 'false' pour plus de sécurité
    }
  }
}

// Instructions pour appliquer ces règles:
// 1. Aller sur https://console.firebase.google.com/project/chasecart-2329d
// 2. Cliquer sur "Firestore Database"
// 3. Aller dans l'onglet "Rules"
// 4. Remplacer le contenu par les règles ci-dessus
// 5. Cliquer sur "Publier"
